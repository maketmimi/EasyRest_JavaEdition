package core;

import core.events.EnRunningStateChangeReason;
import core.events.GeneralEvent;
import core.events.IRunningStateChangedListener;
import core.helpers.fileHelpers.FileHelper;
import java.time.Duration;
import core.events.IGeneralListener;
import core.events.RunningStateChangedEvent;
import core.events.RunningStateChangedEventArgs;
import core.events.SimpleEventArgs;
import java.util.Timer;
import java.util.TimerTask;
import java.util.regex.Pattern;

public class EasyRestStatus {

    private static final String _RecordSeperator  = "[RestStatus]";

    // fields

    private final Timer _tickTimer = new Timer();
    private boolean _isRunning = false;

    private final GeneralEvent _onNumberOfSecondsRemainingChanged = new GeneralEvent();
    private final GeneralEvent _onCurrentPeriodChanged = new GeneralEvent();
    private final RunningStateChangedEvent _onRunningStateChanged = new RunningStateChangedEvent();
    
    private long _NumberOfSecondsRemainig;
    private Period _RestPeriod = null;
    private Period _WorkPeriod = null;
    private Period _CurrentPeriod = null; 

    public void addOnNumberOfSecondsRemainingChangedListener(IGeneralListener listenerToAdd) {
        _onNumberOfSecondsRemainingChanged.addListener(listenerToAdd);
    }

    public void addOnCurrentPeriodChangedListener(IGeneralListener listenerToAdd) {
        _onCurrentPeriodChanged.addListener(listenerToAdd);
    }
    
    public void addOnRunningStateChangedListener(IRunningStateChangedListener listenerToAdd) {
        _onRunningStateChanged.addListener(listenerToAdd);
    }
    
    public long getNumberOfSecondsRemainig() {
        return _NumberOfSecondsRemainig;
    }

    private void setNumberOfSecondsRemaining(long NumberOfSecondsRemainig) {
        
        this._NumberOfSecondsRemainig = NumberOfSecondsRemainig;
        
        _onNumberOfSecondsRemainingChanged.fire(new SimpleEventArgs(this));
        
    }

    public Period getRestPeriod() {
        return _RestPeriod;
    }

    private void setRestPeriod(Period RestPeriod) {
        
        this._RestPeriod = RestPeriod;
        this._RestPeriod.addOnDurationChangedListener(this::periodDurationChanged);
        
    }

    public Period getWorkPeriod() {
        return _WorkPeriod;
    }

    private void setWorkPeriod(Period WorkPeriod) {

        this._WorkPeriod = WorkPeriod;
        this._WorkPeriod.addOnDurationChangedListener(this::periodDurationChanged);
        
    }

    public Period getCurrentPeriod() {
        return _CurrentPeriod;
    }

    public void setCurrentPeriod(Period CurrentPeriod) {

        if (CurrentPeriod != null && (CurrentPeriod == _RestPeriod || CurrentPeriod == _WorkPeriod)) 
        {
            _CurrentPeriod = CurrentPeriod;
            
            _onCurrentPeriodChanged.fire(new SimpleEventArgs(this));
            
            setNumberOfSecondsRemaining(getCurrentPeriod().getDuration().toSeconds());
        }

    }
    
    // methods & constructors
    
    private EasyRestStatus(){
        
        _tickTimer.scheduleAtFixedRate(new TimerTask() {
            @Override
            public void run() {

                tick();

            }
        }, 1000, 1000);
        
    }
    
    public EasyRestStatus(boolean isRunning) {
        
        this();
        
        setWorkPeriod(new Period("Work Period", "It is time for rest", Duration.ofMinutes(20), Period.EnAlarmType.ScreenOverlay, null));
        setRestPeriod(new Period("Rest Period", "It is time for work", Duration.ofMinutes(10), Period.EnAlarmType.ScreenOverlay, null));

        setCurrentPeriod(_WorkPeriod);

        if (isRunning)
            start();
    }

    private EasyRestStatus(Period workPeriod, Period restPeriod, boolean isRunning) {
        
        this();
        
        setWorkPeriod(workPeriod);
        setRestPeriod(restPeriod);
        
        setCurrentPeriod(_WorkPeriod);

        if (isRunning)
            start();
    }
    
    
    private void periodDurationChanged(SimpleEventArgs eventArgs){
        
        if (getCurrentPeriod() == ((Period) eventArgs.getSource()))
            setNumberOfSecondsRemaining(((Period) eventArgs.getSource()).getDuration().toSeconds());
        
    }
    
    private void switchPeriods() {
        
        if (getCurrentPeriod() == _RestPeriod) 
        {
            setCurrentPeriod(_WorkPeriod);
        } 
        else
        {
            setCurrentPeriod(_RestPeriod);
        }
        
    }

    public void pause() {

        pause(EnRunningStateChangeReason.NormalCall);
        
    }

    public void start() {

        start(EnRunningStateChangeReason.NormalCall);

    }
    
    private void start(EnRunningStateChangeReason reason){

        _isRunning = true;
        _onRunningStateChanged.fire(new RunningStateChangedEventArgs(this, reason));
        
    }
    
    private void pause(EnRunningStateChangeReason reason){
        
        _isRunning = false;
        _onRunningStateChanged.fire(new RunningStateChangedEventArgs(this, reason));
   
    }
    
    public boolean isRunning()
    {
        return _isRunning;
    }
    
    // this returns the String of the remining seconds in Minutes:Seconds format
    public String getReminingDurationString() 
    {
        int minutes = (int) getNumberOfSecondsRemainig() / 60;
        int seconds = (int) getNumberOfSecondsRemainig() % 60;
        
        return String.format("%02d:%02d", minutes, seconds);
    }

    public String convertToStringRecord() {
        return String.join(_RecordSeperator, _WorkPeriod.convertToStringRecord(), _RestPeriod.convertToStringRecord());
    }

    public static EasyRestStatus convertStringRecordToStatusObject(String Record, boolean isRunning) {
        
        if (Record == null) return null;

        try 
        {
            String[] arrRecord = Record.split(Pattern.quote(_RecordSeperator));
            
            Period WorkPeriod = Period.convertStringRecordToPeriod(arrRecord[0]);
            Period RestPeriod = Period.convertStringRecordToPeriod(arrRecord[1]);

            if (WorkPeriod == null || RestPeriod == null) {
                return null;
            }

            return new EasyRestStatus(WorkPeriod, RestPeriod, isRunning);
        } 
        catch (Exception any)
        {
            return null;
        }
    }

    private void tick(){
        
        if (!_isRunning) return;
        
        setNumberOfSecondsRemaining(_NumberOfSecondsRemainig - 1);
        
        if (getNumberOfSecondsRemainig() == 0)
        {
            pause(EnRunningStateChangeReason.TickCall);
            
            getCurrentPeriod().showEndAlarm();
            
            switchPeriods();
            
            start(EnRunningStateChangeReason.TickCall);
        }
    }
    
    public boolean saveToFile(String fileName){
        
        return FileHelper.writeRecordToFile(fileName, convertToStringRecord());
        
    }
    
    public static EasyRestStatus loadFromFile(String fileName, boolean isRunning){
        
        return convertStringRecordToStatusObject(FileHelper.readRecordFromFile(fileName), isRunning);
        
    }
    
    public static EasyRestStatus loadFromFileIfAnyOrDefault(String fileName, boolean isRunning){
        
        EasyRestStatus easyRestStatusToReturn = loadFromFile(fileName, isRunning);
        
        if (easyRestStatusToReturn != null)
            return easyRestStatusToReturn;
        else
            return new EasyRestStatus(isRunning);
        
    }
    
}

package core;

import core.alarms.Notification;
import core.alarms.OverlayAlarm;

import java.time.Duration;

import core.events.IGeneralListener;
import core.events.SimpleEventArgs;
import java.util.regex.Pattern;

public class Period {

    public enum EnAlarmType {
        NotificationsOnly,
        SoundOnly,
        ScreenOverlay
    }
    
    private static final String _RecordSeperator  = "[PSep]";

    // class properties
    
    private String _Title;
    private Duration _duration;
    
    private GeneralEvent _onDurationChanged = new GeneralEvent();
    
    private String _EndOfPeriodMessage;
    private EnAlarmType _AlarmType;

    // path to alarm sound, null or white space means Default sound
    private String _AlarmSoundPath;

    public void addOnDurationChangedListener(IGeneralListener listenerToAdd) {
        _onDurationChanged.addListener(listenerToAdd);
    }
    
    public String getAlarmSoundPath() {
        
        if (StringHelpers.isNullOrBlank(_AlarmSoundPath))
            return Configs.defaultAlarmSoundPath;
        else
            return _AlarmSoundPath;
        
    }

    public void setAlarmSoundPath(String _AlarmSoundPath) {
        this._AlarmSoundPath = _AlarmSoundPath;
    }

    public String getTitle() {
        return _Title;
    }

    public void setTitle(String Title) {
        this._Title = Title;
    }

    public Duration getDuration() {
        return _duration;
    }

    public void setDuration(Duration duration) {
        
        this._duration = duration.toSeconds() < 1 ? Duration.ofSeconds(1) : duration;
        _onDurationChanged.fire(new SimpleEventArgs(this));
        
    }

    public String getEndOfPeriodMessage() {
        return _EndOfPeriodMessage;
    }

    public void setEndOfPeriodMessage(String EndOfPeriodMessage) {
        this._EndOfPeriodMessage = EndOfPeriodMessage;
    }

    public EnAlarmType getAlarmType() {
        return _AlarmType;
    }

    public void setAlarmType(EnAlarmType AlarmType) {
        this._AlarmType = AlarmType;
    }

    // class methods and constructors

    // null alarm sound path for default sound
    public Period(String title, String endOfPeriodMessage, Duration duration, EnAlarmType alarmType, String AlarmSoundPath) {
        
        this._Title = title;
        this._EndOfPeriodMessage = endOfPeriodMessage;
        this._duration = duration.toSeconds() < 1 ? Duration.ofSeconds(1) : duration;
        this._AlarmType = alarmType;
        this._AlarmSoundPath = AlarmSoundPath;

    }

    private void showScreenOverlayAlarm() {
        
        OverlayAlarm.showAlarm(getEndOfPeriodMessage(), (short) 10, getAlarmSoundPath());

    }

    private void showNotificationAlarm() throws Exception
    {
        Notification.showNotificationMessage("Alarm", getEndOfPeriodMessage(), TrayIcon.MessageType.INFO, Configs.getAppIconImage(), (short) 20);
    }

    private void playSoundOnlyAlarm() throws Exception
    {
        SoundOnlyAlarm.PlayAlarm((short) 10, getAlarmSoundPath());
    }

    public void showEndAlarm() {
        
        try{
            
            switch (getAlarmType()) {
                case EnAlarmType.NotificationsOnly:
                    showNotificationAlarm();
                    break;
                case EnAlarmType.SoundOnly:
                    playSoundOnlyAlarm();
                    break;
                default:
                    showScreenOverlayAlarm();
                    break;
            }
        }
        catch (Exception any){
            
            showScreenOverlayAlarm();
            
        }
    }

    @Override
    public String toString() {
        return getTitle();
    }

    public String convertToStringRecord() {
        return String.join(_RecordSeperator, getTitle(), getEndOfPeriodMessage(), Long.toString(getDuration().toSeconds()), getAlarmType().toString(), getAlarmSoundPath());
    }

    public static Period convertStringRecordToPeriod(String Record) {
        
        String[] arrRecord = Record.split(Pattern.quote(_RecordSeperator));

        try 
        {
            
            return new Period
            (
                arrRecord[0], 
                arrRecord[1],
                Duration.ofSeconds(Long.parseLong(arrRecord[2])),
                EnAlarmType.valueOf(arrRecord[3]),
                arrRecord[4]
            );
            
        }
        catch (Exception any)  
        {
            return null;
        }
    
    }

}

package core.alarms;

import java.awt.AWTException;
import java.awt.Image;
import java.awt.SystemTray;
import java.awt.TrayIcon;
import java.awt.event.ActionEvent;
import javax.swing.Timer;

public class Notification {

    public static void showNotificationMessage(String title, String message, TrayIcon.MessageType messageType, Image trayIconImage, short showTimeInSeconds) throws UnsupportedOperationException, AWTException
    {
        new Notification(showTimeInSeconds, trayIconImage).Show(title, message, messageType);
    }
    
    private final SystemTray _systemTray;
    private final TrayIcon _trayIcon;
    private final Timer _notificationTimer;
    
    public Notification(short showTimeInSeconds, Image trayIconImage) throws UnsupportedOperationException, AWTException
    {
        _systemTray = SystemTray.getSystemTray();
        
        _trayIcon = new TrayIcon(trayIconImage);
        _trayIcon.setImageAutoSize(true);
        
        _systemTray.add(_trayIcon);
        
        _notificationTimer = new Timer(showTimeInSeconds * 1000, this::notificationTimedOut);
    }

    private void Show(String title, String message, TrayIcon.MessageType messageType){
        
        _trayIcon.displayMessage(title, message, messageType);
        _notificationTimer.start();
        
    }
    
    private void notificationTimedOut(ActionEvent e){

        _notificationTimer.stop();
        _systemTray.remove(_trayIcon);

    }
    
}

package core.events;

import java.util.ArrayList;
import java.util.function.Consumer;


public abstract class SimpleEvent<listenerType, eventArgsType extends SimpleEventArgs>{

    private final ArrayList<listenerType> _registeredListeners = new ArrayList<listenerType>();

    public SimpleEvent() {

    
    }
    
    public void addListener(listenerType listenerToAdd){
        
        if (listenerToAdd != null)
            _registeredListeners.add(listenerToAdd);
        
    }
    
    public boolean removeListener(listenerType listenerToRemove){
        
        if (listenerToRemove != null)
            return _registeredListeners.remove(listenerToRemove);
        
        return false;
    }
    
    public void fire(eventArgsType eventArgs){

        _registeredListeners.forEach(new Consumer<listenerType>(){
            @Override
            public void accept(listenerType t) {

                fireAction(t, eventArgs);

            }
        });
        
    }
    
    protected abstract void fireAction(listenerType listener, eventArgsType eventArgs);
    
}

import java.io.BufferedWriter;

public final class FileHelper {

    public static boolean writeRecordToFile(String fileName, String recordToWrite){
        
        try{

            File parentDir = new File(fileName).getParentFile();
            if (parentDir != null && !parentDir.exists()) parentDir.mkdirs();
            
            BufferedWriter bWriter = new BufferedWriter(new FileWriter(fileName));
            
            bWriter.write(recordToWrite);

            bWriter.close();
            
            return true;
        }
        catch (Exception any){
                              
            return false;

    private void applySettings(){
        
        _appMainWindow.getEasyRestStatus().setCurrentPeriod((Period) _jCmbCurrentPeriod.getSelectedItem());
        
        _appMainWindow.getEasyRestStatus().getWorkPeriod().setDuration(Duration.ofMinutes(((Number)_jNudWorkPeriodDuration.getValue()).longValue()));
        _appMainWindow.getEasyRestStatus().getWorkPeriod().setAlarmType((Period.EnAlarmType) _jCmbWorkPeriodAlarmType.getSelectedItem());
        _appMainWindow.getEasyRestStatus().getWorkPeriod().setAlarmSoundPath(getWorkPeriodAlarmSoundPath());
        
        _appMainWindow.getEasyRestStatus().getRestPeriod().setDuration(Duration.ofMinutes(((Number)_jNudRestPeriodDuration.getValue()).longValue()));
        _appMainWindow.getEasyRestStatus().getRestPeriod().setAlarmType((Period.EnAlarmType) _jCmbRestPeriodAlarmType.getSelectedItem());
        _appMainWindow.getEasyRestStatus().getRestPeriod().setAlarmSoundPath(getRestPeriodAlarmSoundPath());
        
        if (_appMainWindow.getEasyRestStatus().saveToFile(Configs.easyRestStatusFilePath))
        {
            JOptionPane.showMessageDialog(_appMainWindow, "Settings saved sucessfully.", "Done", JOptionPane.INFORMATION_MESSAGE);
        }
        else
        {
            JOptionPane.showMessageDialog(_appMainWindow, "Cannot save settings!.", "ERROR!", JOptionPane.ERROR_MESSAGE);
        }
            
    }

    private void _jBtnDefaultSoundClicked(ActionEvent e){
        
        if (e.getSource() == _jBtnWorkPeriodDefaultAlarmSound)
            setWorkPeriodAlarmSoundPath(Configs.defaultAlarmSoundPath);
        else if (e.getSource() == _jBtnRestPeriodDefaultAlarmSound)
            setRestPeriodAlarmSoundPath(Configs.defaultAlarmSoundPath);
        
    }
    
    private boolean checkAudioFile(String fileName){
        
        if (SoundHelper.isAudioFileSupported(fileName))
        {
            return true;
        }
        else
        {
            JOptionPane.showMessageDialog(_appMainWindow, "Ops... this audio file is not supported!", "Not supported audio file!", JOptionPane.ERROR_MESSAGE);
            return false;
        }
    
    }
    
    private void _jBtnChooseSoundClicked(ActionEvent e){
        
        if (_jFchChooseAlarmSound.showOpenDialog(_appMainWindow) == JFileChooser.APPROVE_OPTION)
        {
            String selectedFilePath = _jFchChooseAlarmSound.getSelectedFile().getAbsolutePath();
            
            if (!checkAudioFile(selectedFilePath))
                return;
            
            if (e.getSource() == _jBtnWorkPeriodChooseSound)
            {
                setWorkPeriodAlarmSoundPath(selectedFilePath);
            }
            else if (e.getSource() == _jBtnRestPeriodChooseSound)
            {
                setRestPeriodAlarmSoundPath(selectedFilePath);
            }
        }
        
    }

import javax.swing.JSpinner;
import javax.swing.SpinnerNumberModel;


public class JNumaricUpDown extends JSpinner {

    public JNumaricUpDown(Number initialValue, Comparable<?> minimum, Comparable<?> maximum, Number stepSize) {
    
        this.setModel(new SpinnerNumberModel(initialValue, minimum, maximum, stepSize));

    }

    @Override
    public void setForeground(Color fg) {

        ((JSpinner.DefaultEditor) this.getEditor()).getTextField().setForeground(fg);

    }
    
    public void setHorizontalAlignment(int horizontalAlignment){
        
        ((JSpinner.DefaultEditor) this.getEditor()).getTextField().setHorizontalAlignment(horizontalAlignment);
        
    }
    
}

package GUI.guiHelpers;

import java.awt.Color;
import java.awt.Font;
import java.awt.GridBagLayout;
import javax.swing.JPanel;
import javax.swing.border.LineBorder;
import javax.swing.border.TitledBorder;


public final class PanelsHelper {

    public static JPanel CreateTitledJPanel(String title){
        
        JPanel createdPanel = new JPanel();
        
        createdPanel.setLayout(new GridBagLayout());
        createdPanel.setOpaque(false);
        LineBorder createdPanelLineBorder = new LineBorder(Color.BLACK, 1);
        TitledBorder createdPanelBorder = new TitledBorder(createdPanelLineBorder, title, TitledBorder.LEFT, TitledBorder.TOP, new Font("Comic Sans MS", Font.BOLD, 30), GUIConfigs.lablesBlue);
        createdPanel.setBorder(createdPanelBorder);
        
        return createdPanel;
    }
    
}
